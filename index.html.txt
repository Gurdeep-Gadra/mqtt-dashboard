            <div class="flex items-center space-x-2">
                <span id="status_dot" class="h-3 w-3 rounded-full bg-amber-500 animate-pulse"></span>

    <script>
        // --- CONFIGURATION ZONE (REPLACE WITH YOUR HIVEMQ DETAILS) ---
        const BROKER_HOST = "e897bc6d6145492e99f4a7f09df8ffd9.s1.eu.hivemq.cloud"; // Do NOT include "wss://" or ports here
        const WEBSOCKET_PORT = 8884;                         // HiveMQ Cloud Secure WebSocket Port
        const MQTT_USER = "Amber_IOT"; 
        const MQTT_PASS = "Amber@123";
        const TOPIC_SUBSCRIBE = "TTW/LAB1";      // Matching your ESP32 subscription path

        // Live ticking header clock helper
        setInterval(() => {
            document.getElementById('live_clock').innerText = new Date().toLocaleTimeString();
        }, 1000);

        const statusText = document.getElementById('status_text');
        const statusDot = document.getElementById('status_dot');
        const statusBar = document.getElementById('status_bar');
        const consoleBox = document.getElementById('console_output');

        // Formulate Secure WebSocket Connection URL string configuration
        const connectionUrl = `wss://${BROKER_HOST}:${WEBSOCKET_PORT}/mqtt`;

        console.log("Connecting to HiveMQ Broker...");
        
        // Initialize MQTT Client options object
        const client = mqtt.connect(connectionUrl, {
            username: MQTT_USER,
            password: MQTT_PASS,
            clientId: 'WebDashboard_' + Math.random().toString(16).substr(2, 8),
            clean: true,
            connectTimeout: 4000,
            reconnectPeriod: 1000,
        });

        // --- MQTT CONNECTION LIFE-CYCLE EVENT HANDLERS ---
        client.on('connect', () => {
            console.log("Connected Successfully over WebSockets!");
            statusBar.className = "bg-gray-900 border border-emerald-500/30 text-emerald-400 px-4 py-3 rounded-lg flex items-center justify-between mb-6 shadow-md";
            statusDot.className = "h-3 w-3 rounded-full bg-emerald-500 shadow-[0_0_8px_rgba(16,185,129,0.6)]";
            statusText.innerText = `Connected: Listening live on [${TOPIC_SUBSCRIBE}]`;
            
            // Subscribe to target string endpoint topic buffer path
            client.subscribe(TOPIC_SUBSCRIBE, (err) => {
                if (err) {
                    appendConsoleLine(`SYSTEM ERROR: Subscribing failed -> ${err.message}`, "text-red-400");
                } else {
                    appendConsoleLine(`SYSTEM LOG: Successfully hooked destination endpoint pipeline subscription.`, "text-gray-500");
                }
            });
        });

        client.on('message', (topic, payload) => {
            const timeStr = new Date().toLocaleTimeString();
            let stringPayload = payload.toString();
            
            // Output layout format styling string parser append routine
            appendConsoleLine(`[${timeStr}] Rx Packet: ${stringPayload}`, "text-emerald-400");
        });

        client.on('error', (err) => {
            console.error("MQTT Client Socket Error Context: ", err);
            appendConsoleLine(`CLIENT SOCKET EXCEPTION: ${err.message}`, "text-red-400");
        });

        client.on('close', () => {
            statusBar.className = "bg-gray-900 border border-red-500/30 text-red-400 px-4 py-3 rounded-lg flex items-center justify-between mb-6 shadow-md";
            statusDot.className = "h-3 w-3 rounded-full bg-red-500";
            statusText.innerText = "Status: Connection Closed / Dropped by HiveMQ.";
        });

        // Safe line append rendering UI subroutines tracking layout visibility bounds
        function appendConsoleLine(text, cssClass) {
            const line = document.createElement('div');
            line.className = cssClass;
            line.innerText = text;
            consoleBox.appendChild(line);
            
            // Automatically push window frame views downward on overflow traces
            consoleBox.scrollTop = consoleBox.scrollHeight;
        }
    </script>
